238static int raft_log_save(raft_t* r) {
239 char path[512];
240 snprintf(path, sizeof(path), "%s/raft_log", r->cfg.data_dir);
241
242 FILE* f = fopen(path, "wb");
243 if (!f) return -1;
244
245 /* 写入 magic + version */
246 uint32_t magic = RAFT_LOG_MAGIC;
247 uint32_t version = RAFT_LOG_VERSION;
248 fwrite(&magic, 4, 1, f);
249 fwrite(&version, 4, 1, f);
250
251 /* 写入快照元数据 */
252 fwrite(&r->snapshot_index, 8, 1, f);
253 fwrite(&r->snapshot_term, 8, 1, f);
254 uint32_t snap_path_len = (uint32_t)strlen(r->snapshot_path);
255 fwrite(&snap_path_len, 4, 1, f);
256 if (snap_path_len > 0) fwrite(r->snapshot_path, 1, snap_path_len, f);
257
258 /* 写入快照和日志大小 */
259 uint64_t snap_size = (uint64_t)r->snapshot_size_bytes;
260 uint64_t log_size = (uint64_t)r->log_size_bytes;
261 fwrite(&snap_size, 8, 1, f);
262 fwrite(&log_size, 8, 1, f);
263
264 /* 写入 current_term */
265 fwrite(&r->current_term, 8, 1, f);
266
267 /* 写入 voted_for */
268 uint32_t voted_len = (uint32_t)strlen(r->voted_for);
269 fwrite(&voted_len, 4, 1, f);
270 fwrite(r->voted_for, 1, voted_len, f);
271
272 /* 写入日志条目数 */
273 uint64_t count = (uint64_t)r->log_count;
274 fwrite(&count, 8, 1, f);
275
276 /* 写入日志条目 */
277 for (size_t i = 0; i < r->log_count; i++) {
278 raft_log_entry_t* e = &r->log[i];
279 fwrite(&e->term, 8, 1, f);
280 fwrite(&e->index, 8, 1, f);
281 fwrite(&e->type, 1, 1, f);
282
283 uint64_t klen = (uint64_t)e->key_len;
284 fwrite(&klen, 8, 1, f);
285 fwrite(e->key, 1, e->key_len, f);
286
287 uint64_t vlen = (uint64_t)e->value_len;
288 fwrite(&vlen, 8, 1, f);
289 if (e->value_len > 0) fwrite(e->value, 1, e->value_len, f);
290 }
291
292 fclose(f);
293 return 0;
294}

296static int raft_log_load(raft_t* r) {
297 char path[512];
298 snprintf(path, sizeof(path), "%s/raft_log", r->cfg.data_dir);
299
300 /* 初始化日志 */
301 r->log_capacity = 1024;
302 r->log = kv_malloc(r->log_capacity * sizeof(raft_log_entry_t));
303 r->log_count = 0;
304
305 FILE* f = fopen(path, "rb");
306 if (!f) {
307 /* 首次启动，无日志 */
308 r->current_term = 0;
309 memset(r->voted_for, 0, sizeof(r->voted_for));
310 r->snapshot_index = 0;
311 r->snapshot_term = 0;
312 r->snapshot_path[0] = '\0';
313 printf("[RAFT] No existing log, starting fresh\n");
314 return 0;
315 }
316
317 /* 获取文件大小 */
318 fseek(f, 0, SEEK_END);
319 long file_size = ftell(f);
320 fseek(f, 0, SEEK_SET);
321
322 if (file_size < 8) { fclose(f); return -1; }
323
324 /* 读取 magic 和 version */
325 uint32_t magic = 0, version = 0;
326 if (fread(&magic, 4, 1, f) != 1) { fclose(f); return -1; }
327 if (fread(&version, 4, 1, f) != 1) { fclose(f); return -1; }
328
329 int is_v2 = (magic == RAFT_LOG_MAGIC && version >= 2);
330 int is_v3 = (magic == RAFT_LOG_MAGIC && version >= 3);
331
332 if (is_v2) {
333 /* --- v2/v3 格式：读取快照元数据 --- */
334 if (fread(&r->snapshot_index, 8, 1, f) != 1) { fclose(f); return -1; }
335 if (fread(&r->snapshot_term, 8, 1, f) != 1) { fclose(f); return -1; }
336
337 uint32_t snap_path_len = 0;
338 if (fread(&snap_path_len, 4, 1, f) != 1) { fclose(f); return -1; }
339 if (snap_path_len > 0 && snap_path_len < sizeof(r->snapshot_path)) {
340 fread(r->snapshot_path, 1, snap_path_len, f);
341 r->snapshot_path[snap_path_len] = '\0';
342 } else {
343 r->snapshot_path[0] = '\0';
344 }
345
346 /* v3: 读取快照和日志大小 */
347 if (is_v3) {
348 uint64_t snap_size = 0, log_size = 0;
349 if (fread(&snap_size, 8, 1, f) != 1) { fclose(f); return -1; }
350 if (fread(&log_size, 8, 1, f) != 1) { fclose(f); return -1; }
351 r->snapshot_size_bytes = (size_t)snap_size;
352 r->log_size_bytes = (size_t)log_size;
353 } else {
354 r->snapshot_size_bytes = 0;
355 r->log_size_bytes = 0;
356 }
357
358 printf("[RAFT] Loaded snapshot metadata: index=%llu, term=%llu\n",
359 (unsigned long long)r->snapshot_index, (unsigned long long)r->snapshot_term);
360 } else {
361 /* v1 格式：无快照，回退到文件开头 */
362 r->snapshot_index = 0;
363 r->snapshot_term = 0;
364 r->snapshot_path[0] = '\0';
365 r->snapshot_size_bytes = 0;
366 r->log_size_bytes = 0;
367 fseek(f, 0, SEEK_SET);
368 }
369
370 /* 读取 current_term */
371 if (fread(&r->current_term, 8, 1, f) != 1) { fclose(f); return -1; }
372
373 /* 读取 voted_for */
374 uint32_t voted_len = 0;
375 if (fread(&voted_len, 4, 1, f) != 1) { fclose(f); return -1; }
376 if (voted_len >= RAFT_NODE_ID_LEN) { fclose(f); return -1; }
377 fread(r->voted_for, 1, voted_len, f);
378 r->voted_for[voted_len] = '\0';
379
380 /* 读取日志条目 */
381 uint64_t count = 0;
382 if (fread(&count, 8, 1, f) != 1) { fclose(f); return -1; }
383 if (count > r->log_capacity) {
384 r->log_capacity = (size_t)count + 1024;
385 void* tmp = kv_realloc(r->log, r->log_capacity * sizeof(raft_log_entry_t));
386 if (!tmp) { fclose(f); return -1; }
387 r->log = (raft_log_entry_t*)tmp;
388 }
389 r->log_count = (size_t)count;
390
391 for (size_t i = 0; i < r->log_count; i++) {
392 raft_log_entry_t* e = &r->log[i];
393 memset(e, 0, sizeof(*e));
394
395 if (fread(&e->term, 8, 1, f) != 1) break;
396 if (fread(&e->index, 8, 1, f) != 1) break;
397 if (fread(&e->type, 1, 1, f) != 1) break;
398
399 uint64_t klen = 0;
400 if (fread(&klen, 8, 1, f) != 1) break;
401 e->key_len = (size_t)klen;
402 e->key = kv_malloc(e->key_len);
403 fread(e->key, 1, e->key_len, f);
404
405 uint64_t vlen = 0;
406 if (fread(&vlen, 8, 1, f) != 1) break;
407 e->value_len = (size_t)vlen;
408 if (vlen > 0) {
409 e->value = kv_malloc(e->value_len);
410 fread(e->value, 1, e->value_len, f);
411 }
412 }
413
414 fclose(f);
415
416 /* 如果 log_size_bytes 为 0（v2 格式），重新计算 */
417 if (r->log_size_bytes == 0) {
418 for (size_t i = 0; i < r->log_count; i++) {
419 r->log_size_bytes += raft_log_entry_size(&r->log[i]);
420 }
421 }
422
423 /* 初始化 commit_index 和 last_applied */
424 /* 如果存在快照，起始索引为快照索引 */
425 uint64_t base_index = r->snapshot_index;
426 if (r->log_count > 0) {
427 r->commit_index = r->log[r->log_count - 1].index;
428 } else {
429 r->commit_index = base_index;
430 }
431 r->last_applied = base_index; /* 快照数据已经 applied，只需应用后续日志 */
432
433 printf("[RAFT] Loaded log: term=%llu, entries=%zu, snapshot_idx=%llu, commit=%llu\n",
434 (unsigned long long)r->current_term, r->log_count,
435 (unsigned long long)r->snapshot_index, (unsigned long long)r->commit_index);
436 return 0;
437}

545 size_t off = 14 + id_len;
546 uint32_t host_len = leader_host ? (uint32_t)strlen(leader_host) : 0;
547 write_u32(buf + off, host_len); off += 4;
548 if (host_len > 0) memcpy(buf + off, leader_host, host_len);
549 off += host_len;
550 write_u32(buf + off, (uint32_t)leader_port); off += 4;
551
552 uint32_t total = (uint32_t)off;
553 send(fd, (const char*)&total, 4, 0);
554 send(fd, (const char*)buf, (int)off, 0);
555 return 0;
556}

562static SOCKET raft_connect_peer(raft_peer_t* peer) {
563 SOCKET fd = socket(AF_INET, SOCK_STREAM, 0);
564 if (fd == INVALID_SOCKET) return INVALID_SOCKET;
565
566 struct sockaddr_in addr;
567 memset(&addr, 0, sizeof(addr));
568 addr.sin_family = AF_INET;
569 addr.sin_port = htons((unsigned short)peer->port);
570 addr.sin_addr.s_addr = inet_addr(peer->host);
571
572 /* 设置超时 */
573 int timeout = RAFT_RPC_TIMEOUT_MS;
574 setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, (const char*)&timeout, sizeof(timeout));
575 setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, (const char*)&timeout, sizeof(timeout));
576
577 if (connect(fd, (struct sockaddr*)&addr, sizeof(addr)) == SOCKET_ERROR) {
578 close_socket(fd);
579 return INVALID_SOCKET;
580 }
581 return fd;
582}

588static void raft_become_follower(raft_t* r, uint64_t term) {
589 r->role = RAFT_FOLLOWER;
590 r->current_term = term;
591 memset(r->voted_for, 0, sizeof(r->voted_for));
592 printf("[RAFT] %s -> FOLLOWER (term=%llu)\n", r->cfg.node_id,
593 (unsigned long long)term);
594}

596static void raft_become_candidate(raft_t* r) {
597 r->role = RAFT_CANDIDATE;
598 r->current_term++;
599 r->votes_received = 1; /* vote for self */
600 snprintf(r->voted_for, sizeof(r->voted_for), "%s", r->cfg.node_id);
601 r->election_timeout_ms = RAFT_ELECTION_TIMEOUT_MIN_MS +
602 (rand() % (RAFT_ELECTION_TIMEOUT_MAX_MS - RAFT_ELECTION_TIMEOUT_MIN_MS));
603 printf("[RAFT] %s -> CANDIDATE (term=%llu, timeout=%llums)\n",
604 r->cfg.node_id, (unsigned long long)r->current_term,
605 (unsigned long long)r->election_timeout_ms);
606}

608static void raft_become_leader(raft_t* r) {
609 r->role = RAFT_LEADER;
610 int self_idx = raft_self_index(r);
611 printf("[RAFT] %s -> LEADER (term=%llu)\n", r->cfg.node_id,
612 (unsigned long long)r->current_term);
613
614 /* 初始化 leader 状态 */
615 uint64_t last_log_idx;
616 if (r->log_count > 0) {
617 last_log_idx = r->log[r->log_count - 1].index;
618 } else if (r->snapshot_index > 0) {
619 last_log_idx = r->snapshot_index;
620 } else {
621 last_log_idx = 0;
622 }
623 for (int i = 0; i < r->cfg.num_peers; i++) {
624 r->next_index[i] = last_log_idx + 1;
625 r->match_index[i] = 0;
626 }
627 r->match_index[self_idx] = last_log_idx;
628
629 /* 更新 leader peer 信息 */
630 snprintf(r->leader_peer.id, sizeof(r->leader_peer.id), "%s", r->cfg.node_id);
631 r->leader_peer = r->cfg.peers[self_idx];
632 r->leader_known = 1;
633
634 /* 立即发送心跳 */
635 r->last_election_ms = 0; /* 触发立即发送 */
636}

638/* 检查自身日志是否至少和 candidate 一样新 */
639static int raft_log_is_up_to_date(raft_t* r, uint64_t last_idx, uint64_t last_term) {
640 if (r->log_count == 0) {
641 /* 日志为空（可能已被快照截断），使用快照信息 */
642 if (r->snapshot_index > 0) {
643 if (last_term != r->snapshot_term) return last_term > r->snapshot_term;
644 return last_idx >= r->snapshot_index;
645 }
646 return 1; /* 无日志无快照，接受任何 */
647 }
648 uint64_t my_last_term = r->log[r->log_count - 1].term;
649 uint64_t my_last_idx = r->log[r->log_count - 1].index;
650 if (last_term != my_last_term) return last_term > my_last_term;
651 return last_idx >= my_last_idx;
652}

654/* 将已提交的日志应用到状态机 */
655static void raft_apply_committed(raft_t* r) {
656 while (r->last_applied < r->commit_index) {
657 uint64_t idx = r->last_applied + 1;
658
659 /* 查找日志条目 */
660 raft_log_entry_t* e = NULL;
661 for (size_t i = 0; i < r->log_count; i++) {
662 if (r->log[i].index == idx) {
663 e = &r->log[i];
664 break;
665 }
666 }
667
668 if (e && r->apply_fn) {
669 raft_entry_t entry;
670 entry.term = e->term;
671 entry.index = e->index;
672 entry.type = e->type;
673 entry.key = e->key;
674 entry.key_len = e->key_len;
675 entry.value = e->value;
676 entry.value_len = e->value_len;
677 r->apply_fn(r->state_machine, &entry);
678 }
679
680 r->last_applied = idx;
681 }

684/* 处理 RequestVote */
685static void raft_handle_vote_request(raft_t* r, SOCKET fd, uint8_t* data, size_t data_len) {
686 if (data_len < 21) return;
687 size_t off = 0;
688 uint64_t term = read_u64(data + off); off += 8;
689 uint32_t id_len = read_u32(data + off); off += 4;
690 if (off + id_len > data_len) return;
691 char candidate_id[RAFT_NODE_ID_LEN];
692 memcpy(candidate_id, data + off, id_len);
693 candidate_id[id_len] = '\0';
694 off += id_len;
695 if (off + 16 > data_len) return;
696 uint64_t last_log_idx = read_u64(data + off); off += 8;
697 uint64_t last_log_term = read_u64(data + off); off += 8;
698
699 MUTEX_LOCK(&r->mutex);
700
701 int grant = 0;
702 if (term < r->current_term) {
703 /* 拒绝：term 太旧 */
704 } else {
705 if (term > r->current_term) {
706 raft_become_follower(r, term);
707 }
708 /* 可以投票：尚未投票给其他人，且 candidate 日志足够新 */
709 if ((r->voted_for[0] == '\0' || strcmp(r->voted_for, candidate_id) == 0) &&
710 raft_log_is_up_to_date(r, last_log_idx, last_log_term)) {
711 grant = 1;
712 snprintf(r->voted_for, sizeof(r->voted_for), "%s", candidate_id);
713 r->last_heartbeat_ms = thread_time_ms(); /* 重置选举计时器 */
714 }
715 }
716
717 MUTEX_UNLOCK(&r->mutex);
718
719 rpc_send_vote_response(fd, r->current_term, grant);
720 if (grant) {
721 raft_log_save(r);
722 printf("[RAFT] Voted for %s (term=%llu)\n", candidate_id, (unsigned long long)term);
723 }
724}

726/* 处理 RequestVote 响应 */
727static void raft_handle_vote_response(raft_t* r, uint8_t* data, size_t data_len) {
728 if (data_len < 10) return;
729 uint64_t term = read_u64(data + 1);
730 int granted = data[9] != 0;
731
732 MUTEX_LOCK(&r->mutex);
733
734 if (r->role != RAFT_CANDIDATE) {
735 MUTEX_UNLOCK(&r->mutex);
736 return;
737 }
738
739 if (term > r->current_term) {
740 raft_become_follower(r, term);
741 raft_log_save(r);
742 MUTEX_UNLOCK(&r->mutex);
743 return;
744 }
745
746 if (granted) {
747 r->votes_received++;
748 printf("[RAFT] Got vote: %d/%d\n", r->votes_received, r->cfg.num_peers);
749 if (r->votes_received > r->cfg.num_peers / 2) {
750 raft_become_leader(r);
751 }
752 }
753
754 MUTEX_UNLOCK(&r->mutex);
755}

757/* 处理 AppendEntries */
758static void raft_handle_append_request(raft_t* r, SOCKET fd, uint8_t* data, size_t data_len) {
759 size_t off = 0;
760 if (off + 1 > data_len) return;
761 off++; /* skip type */
762 if (off + 8 > data_len) return;
763 uint64_t term = read_u64(data + off); off += 8;
764 uint32_t id_len = read_u32(data + off); off += 4;
765 if (off + id_len > data_len) return;
766 char leader_id[RAFT_NODE_ID_LEN];
767 memcpy(leader_id, data + off, id_len);
768 leader_id[id_len] = '\0';
769 off += id_len;
770 if (off + 32 > data_len) return;
771 uint64_t prev_idx = read_u64(data + off); off += 8;
772 uint64_t prev_term = read_u64(data + off); off += 8;
773 uint64_t n_entries = read_u64(data + off); off += 8;
774 uint64_t leader_commit = read_u64(data + off); off += 8;
775
776 MUTEX_LOCK(&r->mutex);
777
778 int success = 0;
779 uint64_t last_idx = prev_idx;
780
781 /* 1. 如果 term < currentTerm，拒绝 */
782 if (term < r->current_term) {
783 /* 获取 last log index（优先日志，否则快照） */
784 if (r->log_count > 0) {
785 last_idx = r->log[r->log_count - 1].index;
786 } else if (r->snapshot_index > 0) {
787 last_idx = r->snapshot_index;
788 }
789 MUTEX_UNLOCK(&r->mutex);
790 rpc_send_append_response(fd, r->current_term, 0, last_idx, 0);
791 return;
792 }
793
794 /* 2. 如果是有效 leader，更新状态 */
795 if (term > r->current_term) {
796 raft_become_follower(r, term);
797 } else {
798 /* 如果收到同一 term 的不同 leader 的 AppendEntries，接受他 */
799 if (r->role == RAFT_CANDIDATE) {
800 raft_become_follower(r, term);
801 }
802 }
803
804 r->last_heartbeat_ms = thread_time_ms();
805 r->role = RAFT_FOLLOWER;
806
807 /* 更新 leader 信息 */
808 {
809 int idx = raft_find_peer(r, leader_id);
810 if (idx >= 0) {
811 r->leader_peer = r->cfg.peers[idx];
812 r->leader_known = 1;
813 }
814 }
815
816 /* 3. 检查 prevLogIndex/prevLogTerm */
817 if (prev_idx > 0) {
818 int found = 0;
819
820 /* 特殊处理：prev_idx 落在快照范围内 */
821 if (r->snapshot_index > 0 && prev_idx <= r->snapshot_index) {
822 if (prev_idx == r->snapshot_index) {
823 /* prev_idx 正好是快照的最后索引，检查 term 是否匹配 */
824 if (prev_term == r->snapshot_term) {
825 found = 1;
826 }
827 } else if (prev_idx < r->snapshot_index) {
828 /* prev_idx 在快照之前，快照已包含这些数据，接受 */
829 found = 1;
830 }
831 }
832
833 if (!found) {
834 for (size_t i = 0; i < r->log_count; i++) {
835 if (r->log[i].index == prev_idx) {
836 if (r->log[i].term == prev_term) {
837 found = 1;
838 }
839 break;
840 }
841 }
842 }
843
844 if (!found && prev_idx != 0) {
845 /* 判断是否需要快照：prev_idx 在快照之前但 term 不匹配 */
846 int need_snap = (r->snapshot_index > 0 && prev_idx <= r->snapshot_index) ? 1 : 0;
847
848 /* 冲突：删除冲突及之后的条目 */
849 size_t new_count = 0;
850 for (size_t i = 0; i < r->log_count; i++) {
851 if (r->log[i].index < prev_idx) {
852 new_count = i + 1;
853 }
854 }
855 /* 释放被删除的条目并更新 log_size_bytes */
856 for (size_t i = new_count; i < r->log_count; i++) {
857 r->log_size_bytes -= raft_log_entry_size(&r->log[i]);
858 kv_free(r->log[i].key);
859 kv_free(r->log[i].value);
860 }
861 r->log_count = new_count;
862 last_idx = r->log_count > 0 ? r->log[r->log_count - 1].index : r->snapshot_index;
863 MUTEX_UNLOCK(&r->mutex);
864 raft_log_save(r);
865 rpc_send_append_response(fd, r->current_term, 0, last_idx, need_snap);
866 return;
867 }
868 }
869
870 /* 4. 追加新条目 */
871 for (uint64_t i = 0; i < n_entries; i++) {
872 if (off + 25 > data_len) break;
873 uint64_t e_term = read_u64(data + off); off += 8;
874 uint64_t e_index = read_u64(data + off); off += 8;
875 uint8_t e_type = data[off++];
876 uint64_t key_len = read_u64(data + off); off += 8;
877 if (off + key_len > data_len) break;
878 char* key = kv_malloc((size_t)key_len);
879 memcpy(key, data + off, (size_t)key_len);
880 off += (size_t)key_len;
881 uint64_t val_len = read_u64(data + off); off += 8;
882 char* val = NULL;
883 if (val_len > 0) {
884 if (off + val_len > data_len) { kv_free(key); break; }
885 val = kv_malloc((size_t)val_len);
886 if (!val) { kv_free(key); break; }
887 memcpy(val, data + off, (size_t)val_len);
888 off += (size_t)val_len;
889 }
890
891 /* 跳过已包含在快照中的条目 */
892 if (r->snapshot_index > 0 && e_index <= r->snapshot_index) {
893 kv_free(key);
894 kv_free(val);
895 last_idx = e_index;
896 continue;
897 }
898
899 /* 检查是否与现有日志冲突 */
900 int found_existing = 0;
901 for (size_t j = 0; j < r->log_count; j++) {
902 if (r->log[j].index == e_index) {
903 if (r->log[j].term != e_term) {
904 /* 冲突：删除此条目及之后 */
905 for (size_t k = j; k < r->log_count; k++) {
906 r->log_size_bytes -= raft_log_entry_size(&r->log[k]);
907 kv_free(r->log[k].key);
908 kv_free(r->log[k].value);
909 }
910 r->log_count = j;
911 } else {
912 /* 已存在且匹配，跳过 */
913 kv_free(key);
914 kv_free(val);
915 key = NULL;
916 }
917 found_existing = 1;
918 break;
919 }
920 }
921 (void)found_existing;
922
923 if (key) {
924 /* 扩容 */
925 if (r->log_count >= r->log_capacity) {
926 r->log_capacity *= 2;
927 void* tmp = kv_realloc(r->log, r->log_capacity * sizeof(raft_log_entry_t));
928 if (!tmp) { kv_free(key); kv_free(val); break; }
929 r->log = (raft_log_entry_t*)tmp;
930 }
931 raft_log_entry_t* ne = &r->log[r->log_count++];
932 ne->term = e_term;
933 ne->index = e_index;
934 ne->type = e_type;
935 ne->key = key;
936 ne->key_len = (size_t)key_len;
937 ne->value = val;
938 ne->value_len = val ? (size_t)val_len : 0;
939 r->log_size_bytes += raft_log_entry_size(ne);
940 }
941
942 last_idx = e_index;
943 }
944
945 /* 5. 更新 commitIndex */
946 if (leader_commit > r->commit_index) {
947 uint64_t last_log = r->log_count > 0 ? r->log[r->log_count - 1].index : 0;
948 r->commit_index = leader_commit < last_log ? leader_commit : last_log;
949 }
950
951 success = 1;
952 MUTEX_UNLOCK(&r->mutex);
953
954 raft_log_save(r);
955 rpc_send_append_response(fd, r->current_term, success, last_idx, 0);
956
957 /* apply committed */
958 MUTEX_LOCK(&r->mutex);
959 raft_apply_committed(r);
960 MUTEX_UNLOCK(&r->mutex);
961}

963/* 处理 AppendEntries 响应 */
964static void raft_handle_append_response(raft_t* r, int peer_index, uint8_t* data, size_t data_len) {
965 if (data_len < 18) return;
966 uint64_t term = read_u64(data + 1);
967 int success = data[9] != 0;
968 uint64_t last_idx = read_u64(data + 10);
969 int need_snapshot = (data_len >= 19) ? (data[18] != 0) : 0;
970
971 MUTEX_LOCK(&r->mutex);
972
973 if (r->role != RAFT_LEADER) {
974 MUTEX_UNLOCK(&r->mutex);
975 return;
976 }
977
978 if (term > r->current_term) {
979 raft_become_follower(r, term);
980 raft_log_save(r);
981 MUTEX_UNLOCK(&r->mutex);
982 return;
983 }
984
985 /* 更新指定 peer 的 match_index 和 next_index */
986 if (peer_index >= 0 && peer_index < r->cfg.num_peers) {
987 if (success) {
988 if (last_idx > r->match_index[peer_index]) {
989 r->match_index[peer_index] = last_idx;
990 r->next_index[peer_index] = last_idx + 1;
991 }
992 } else {
993 /* Follower 发送了 need_snapshot 标志：立即发送快照 */
994 if (need_snapshot && r->snapshot_index > 0) {
995 printf("[RAFT] Follower %s needs snapshot (snapshot_index=%llu)\n",
996 r->cfg.peers[peer_index].id, (unsigned long long)r->snapshot_index);
997 MUTEX_UNLOCK(&r->mutex);
998 raft_leader_send_snapshot(r, peer_index);
999 return;
1000 }
1001
1002 /* 使用 last_idx 快速跳过（而非逐个递减） */
1003 if (last_idx > 0 && last_idx < r->next_index[peer_index]) {
1004 r->next_index[peer_index] = last_idx + 1;
1005 if (r->next_index[peer_index] < 1) r->next_index[peer_index] = 1;
1006 } else if (r->next_index[peer_index] > 1) {
1007 r->next_index[peer_index]--;
1008 }
1009 }
1010 }
1011
1012 /* 更新 commitIndex */
1013 uint64_t last_log_idx = r->log_count > 0 ? r->log[r->log_count - 1].index : 0;
1014 for (uint64_t n = r->commit_index + 1; n <= last_log_idx; n++) {
1015 /* 查找该索引的 term */
1016 uint64_t n_term = 0;
1017 for (size_t j = 0; j < r->log_count; j++) {
1018 if (r->log[j].index == n) { n_term = r->log[j].term; break; }
1019 }
1020 if (n_term != r->current_term) continue;
1021
1022 int count = 1; /* 自己的票 */
1023 for (int i = 0; i < r->cfg.num_peers; i++) {
1024 if (i == raft_self_index(r)) continue;
1025 if (r->match_index[i] >= n) count++;
1026 }
1027 if (count > r->cfg.num_peers / 2) {
1028 r->commit_index = n;
1029 }
1030 }
1031
1032 MUTEX_UNLOCK(&r->mutex);
1033
1034 /* apply committed */
1035 MUTEX_LOCK(&r->mutex);
1036 raft_apply_committed(r);
1037 MUTEX_UNLOCK(&r->mutex);
1038}

1040/* Leader 发送心跳 / 日志复制 */
1041static void raft_leader_send_heartbeat(raft_t* r) {
1042 int self_idx = raft_self_index(r);
1043 uint64_t prev_idx, prev_term;
1044
1045 for (int i = 0; i < r->cfg.num_peers; i++) {
1046 if (i == self_idx) continue;
1047
1048 /* 准备发送的日志条目 */
1049 uint64_t next = r->next_index[i];
1050
1051 /* 如果 follower 落后太多（next_index <= snapshot_index），发送快照 */
1052 if (r->snapshot_index > 0 && next <= r->snapshot_index) {
1053 raft_leader_send_snapshot(r, i);
1054 continue;
1055 }
1056
1057 prev_idx = next - 1;
1058 prev_term = 0;
1059
1060 if (prev_idx > 0) {
1061 /* 如果 prev_idx 正好是快照索引，使用快照的 term */
1062 if (r->snapshot_index > 0 && prev_idx == r->snapshot_index) {
1063 prev_term = r->snapshot_term;
1064 } else {
1065 for (size_t j = 0; j < r->log_count; j++) {
1066 if (r->log[j].index == prev_idx) {
1067 prev_term = r->log[j].term;
1068 break;
1069 }
1070 }
1071 }
1072 }
1073
1074 /* 收集要发送的日志 */
1075 raft_log_entry_t send_entries[128];
1076 int n_send = 0;
1077 uint64_t send_idx = next;
1078 for (size_t j = 0; j < r->log_count && n_send < 128; j++) {
1079 if (r->log[j].index >= send_idx) {
1080 send_entries[n_send++] = r->log[j];
1081 send_idx++;
1082 }
1083 }
1084
1085 SOCKET fd = raft_connect_peer(&r->cfg.peers[i]);
1086 if (fd == INVALID_SOCKET) {
1087 continue;
1088 }
1089
1090 rpc_send_append_request(fd, r->current_term, r->cfg.node_id,
1091 prev_idx, prev_term, send_entries, n_send,
1092 r->commit_index);
1093
1094 /* 读取响应 */
1095 uint32_t resp_len = 0;
1096 int ret = recv(fd, (char*)&resp_len, 4, 0);
1097 if (ret == 4) {
1098 uint8_t resp_buf[256];
1099 int total = recv(fd, (char*)resp_buf, (int)resp_len, 0);
1100 if (total == (int)resp_len) {
1101 raft_handle_append_response(r, i, resp_buf, resp_len);
1102 }
1103 }
1104 close_socket(fd);
1105 }
1106}

1108/* Candidate 发起选举 */
1109static void raft_candidate_start_election(raft_t* r) {
1110 r->votes_received = 1; /* vote for self */
1111 r->election_timeout_ms = RAFT_ELECTION_TIMEOUT_MIN_MS +
1112 (rand() % (RAFT_ELECTION_TIMEOUT_MAX_MS - RAFT_ELECTION_TIMEOUT_MIN_MS));
1113 r->last_heartbeat_ms = thread_time_ms(); /* 重置计时器 */
1114
1115 uint64_t last_log_idx;
1116 uint64_t last_log_term;
1117 if (r->log_count > 0) {
1118 last_log_idx = r->log[r->log_count - 1].index;
1119 last_log_term = r->log[r->log_count - 1].term;
1120 } else if (r->snapshot_index > 0) {
1121 last_log_idx = r->snapshot_index;
1122 last_log_term = r->snapshot_term;
1123 } else {
1124 last_log_idx = 0;
1125 last_log_term = 0;
1126 }
1127
1128 int self_idx = raft_self_index(r);
1129 for (int i = 0; i < r->cfg.num_peers; i++) {
1130 if (i == self_idx) continue;
1131
1132 SOCKET fd = raft_connect_peer(&r->cfg.peers[i]);
1133 if (fd == INVALID_SOCKET) continue;
1134
1135 rpc_send_vote_request(fd, r->current_term, r->cfg.node_id,
1136 last_log_idx, last_log_term);
1137
1138 /* 读取响应 */
1139 uint32_t resp_len = 0;
1140 int ret = recv(fd, (char*)&resp_len, 4, 0);
1141 if (ret == 4) {
1142 uint8_t resp_buf[64];
1143 int total = recv(fd, (char*)resp_buf, (int)resp_len, 0);
1144 if (total == (int)resp_len) {
1145 raft_handle_vote_response(r, resp_buf, resp_len);
1146 }
1147 }
1148 close_socket(fd);
1149 }
1150
1151 raft_log_save(r);
1152}

1154/* 处理客户端 Propose 请求 */
1155static void raft_handle_propose_request(raft_t* r, SOCKET fd, uint8_t* data, size_t data_len) {
1156 size_t off = 0;
1157 if (off + 1 > data_len) return;
1158 uint8_t cmd_type = data[off++];
1159 if (off + 8 > data_len) return;
1160 uint64_t key_len = read_u64(data + off); off += 8;
1161 if (off + key_len > data_len) return;
1162 char* key = kv_malloc((size_t)key_len + 1);
1163 memcpy(key, data + off, (size_t)key_len);
1164 key[key_len] = '\0';
1165 off += (size_t)key_len;
1166 if (off + 8 > data_len) { kv_free(key); return; }
1167 uint64_t val_len = read_u64(data + off); off += 8;
1168 char* val = NULL;
1169 if (val_len > 0) {
1170 if (off + val_len > data_len) { kv_free(key); return; }
1171 val = kv_malloc((size_t)val_len);
1172 memcpy(val, data + off, (size_t)val_len);
1173 off += (size_t)val_len;
1174 }
1175
1176 MUTEX_LOCK(&r->mutex);
1177
1178 /* 只有 Leader 可以处理 propose */
1179 if (r->role != RAFT_LEADER) {
1180 if (r->leader_known) {
1181 rpc_send_propose_response(fd, r->current_term, 0,
1182 r->leader_peer.id, r->leader_peer.host,
1183 r->leader_peer.port);
1184 } else {
1185 rpc_send_propose_response(fd, r->current_term, 0, NULL, NULL, 0);
1186 }
1187 kv_free(key);
1188 kv_free(val);
1189 MUTEX_UNLOCK(&r->mutex);
1190 return;
1191 }
1192
1193 /* 扩容日志 */
1194 if (r->log_count >= r->log_capacity) {
1195 r->log_capacity *= 2;
1196 void* tmp = kv_realloc(r->log, r->log_capacity * sizeof(raft_log_entry_t));
1197 if (!tmp) { MUTEX_UNLOCK(&r->mutex); return; }
1198 r->log = (raft_log_entry_t*)tmp;
1199 }
1200
1201 uint64_t new_idx = r->log_count > 0 ? r->log[r->log_count - 1].index + 1 : 1;
1202 raft_log_entry_t* ne = &r->log[r->log_count++];
1203 ne->term = r->current_term;
1204 ne->index = new_idx;
1205 ne->type = cmd_type;
1206 ne->key = key;
1207 ne->key_len = (size_t)key_len;
1208 ne->value = val;
1209 ne->value_len = val ? (size_t)val_len : 0;
1210
1211 r->log_size_bytes += raft_log_entry_size(ne);
1212
1213 /* 立即尝试复制 */
1214 r->match_index[raft_self_index(r)] = new_idx;
1215
1216 MUTEX_UNLOCK(&r->mutex);
1217 raft_log_save(r);
1218
1219 /* 发送 AppendEntries 给所有 follower */
1220 raft_leader_send_heartbeat(r);
1221
1222 /* 等待提交 */
1223 int max_wait = 100; /* 最多等待 100ms */
1224 while (max_wait-- > 0) {
1225 MUTEX_LOCK(&r->mutex);
1226 if (r->commit_index >= new_idx) {
1227 raft_apply_committed(r);
1228 MUTEX_UNLOCK(&r->mutex);
1229 rpc_send_propose_response(fd, r->current_term, 1, NULL, NULL, 0);
1230 return;
1231 }
1232 MUTEX_UNLOCK(&r->mutex);
1233 thread_sleep_ms(1);
1234 }
1235
1236 rpc_send_propose_response(fd, r->current_term, 0, NULL, NULL, 0);
1237}

1243#ifdef _WIN32
1244static DWORD WINAPI raft_event_loop(LPVOID arg) {
1245#else
1246static void* raft_event_loop(void* arg) {
1247#endif
1248 raft_t* r = (raft_t*)arg;
1249 r->running = 1;
1250
1251 printf("[RAFT] %s event loop started\n", r->cfg.node_id);
1252
1253 while (r->running) {
1254 /* 使用 select 等待连接或超时 */
1255 fd_set read_fds;
1256 FD_ZERO(&read_fds);
1257 FD_SET(r->listen_fd, &read_fds);
1258
1259 struct timeval tv;
1260 tv.tv_sec = 0;
1261 tv.tv_usec = 50000; /* 50ms 轮询 */
1262
1263 int ret = select((int)r->listen_fd + 1, &read_fds, NULL, NULL, &tv);
1264 if (ret < 0) {
1265 if (
1266#ifdef _WIN32
1267 WSAGetLastError() == WSAEINTR
1268#else
1269 errno == EINTR
1270#endif
1271 ) continue;
1272 break;
1273 }
1274
1275 uint64_t now = thread_time_ms();
1276
1277 MUTEX_LOCK(&r->mutex);
1278
1279 /* --- 选举超时检查 --- */
1280 if (r->role != RAFT_LEADER) {
1281 if (now - r->last_heartbeat_ms >= r->election_timeout_ms) {
1282 raft_become_candidate(r);
1283 raft_log_save(r);
1284 }
1285 }
1286
1287 /* --- Leader 心跳 --- */
1288 if (r->role == RAFT_LEADER) {
1289 if (now - r->last_election_ms >= RAFT_HEARTBEAT_INTERVAL_MS) {
1290 r->last_election_ms = now;
1291 MUTEX_UNLOCK(&r->mutex);
1292 raft_leader_send_heartbeat(r);
1293 MUTEX_LOCK(&r->mutex);
1294 }
1295
1296 /* 自动触发快照（周期性：日志积累超过阈值即触发） */
1297 if (r->log_count >= RAFT_SNAPSHOT_LOG_THRESHOLD ||
1298 r->log_size_bytes >= RAFT_SNAPSHOT_LOG_SIZE_THRESHOLD) {
1299 /* 避免频繁快照：至少等 5 秒，或日志增长超过上次快照的 50% */
1300 static uint64_t last_snapshot_ms = 0;
1301 uint64_t now2 = thread_time_ms();
1302 int should_snap = 0;
1303 if (last_snapshot_ms == 0 || (now2 - last_snapshot_ms) > 5000) {
1304 should_snap = 1;
1305 } else if (r->snapshot_index > 0 &&
1306 r->log_count > 0 &&
1307 r->log[r->log_count - 1].index - r->snapshot_index > RAFT_SNAPSHOT_LOG_THRESHOLD / 2) {
1308 should_snap = 1;
1309 }
1310 if (should_snap) {
1311 last_snapshot_ms = now2;
1312 MUTEX_UNLOCK(&r->mutex);
1313 raft_snapshot_create(r);
1314 MUTEX_LOCK(&r->mutex);
1315 }
1316 }
1317 }
1318
1319 /* --- Candidate 发起选举 --- */
1320 if (r->role == RAFT_CANDIDATE) {
1321 /* 选举在超时触发时已经发起 */
1322 if (now - r->last_heartbeat_ms >= r->election_timeout_ms) {
1323 MUTEX_UNLOCK(&r->mutex);
1324 raft_candidate_start_election(r);
1325 MUTEX_LOCK(&r->mutex);
1326 }
1327 }
1328
1329 MUTEX_UNLOCK(&r->mutex);
1330
1331 /* --- 接受新连接 --- */
1332 if (FD_ISSET(r->listen_fd, &read_fds)) {
1333 struct sockaddr_in client_addr;
1334 socklen_t addr_len = sizeof(client_addr);
1335 SOCKET client_fd = accept(r->listen_fd, (struct sockaddr*)&client_addr, &addr_len);
1336 if (client_fd != INVALID_SOCKET) {
1337 /* 读取消息 */
1338 uint32_t msg_len = 0;
1339 int n = recv(client_fd, (char*)&msg_len, 4, 0);
1340 if (n == 4 && msg_len > 0 && msg_len < RAFT_MAX_MSG_SIZE) {
1341 uint8_t* msg = kv_malloc(msg_len);
1342 int total = recv(client_fd, (char*)msg, (int)msg_len, 0);
1343 if (total == (int)msg_len && msg_len > 0) {
1344 switch (msg[0]) {
1345 case RAFT_RPC_VOTE_REQ:
1346 raft_handle_vote_request(r, client_fd, msg, msg_len);
1347 break;
1348 case RAFT_RPC_APPEND_REQ:
1349 raft_handle_append_request(r, client_fd, msg, msg_len);
1350 break;
1351 case RAFT_RPC_PROPOSE_REQ:
1352 raft_handle_propose_request(r, client_fd, msg, msg_len);
1353 break;
1354 case RAFT_RPC_SNAPSHOT_REQ:
1355 raft_handle_snapshot_request(r, client_fd, msg, msg_len);
1356 break;
1357 case RAFT_RPC_SNAPSHOT_CHUNK:
1358 raft_handle_snapshot_chunk(r, client_fd, msg, msg_len);
1359 break;
1360 default:
1361 break;
1362 }
1363 }
1364 kv_free(msg);
1365 }
1366 close_socket(client_fd);
1367 }
1368 }
1369 }
1370
1371 printf("[RAFT] %s event loop stopped\n", r->cfg.node_id);
1372 return 0;
1373}

1515int raft_propose(raft_t* r, uint8_t type, const char* key, size_t key_len,
1516 const char* value, size_t value_len) {
1517 if (!r || !key || key_len == 0) return -1;
1518
1519 MUTEX_LOCK(&r->mutex);
1520
1521 if (r->role != RAFT_LEADER) {
1522 if (r->leader_known) {
1523 /* 转发到 Leader */
1524 SOCKET fd = raft_connect_peer(&r->leader_peer);
1525 if (fd == INVALID_SOCKET) {
1526 MUTEX_UNLOCK(&r->mutex);
1527 return -1;
1528 }
1529
1530 uint8_t buf[4096];
1531 size_t off = 0;
1532 buf[off++] = RAFT_RPC_PROPOSE_REQ;
1533 buf[off++] = type;
1534 write_u64(buf + off, (uint64_t)key_len); off += 8;
1535 memcpy(buf + off, key, key_len); off += key_len;
1536 write_u64(buf + off, (uint64_t)value_len); off += 8;
1537 if (value_len > 0) memcpy(buf + off, value, value_len);
1538 off += value_len;
1539
1540 uint32_t total = (uint32_t)off;
1541 send(fd, (const char*)&total, 4, 0);
1542 send(fd, (const char*)buf, (int)off, 0);
1543
1544 /* 读取响应 */
1545 uint32_t resp_len = 0;
1546 int ret = recv(fd, (char*)&resp_len, 4, 0);
1547 int result = -1;
1548 if (ret == 4 && resp_len > 0) {
1549 uint8_t* resp = kv_malloc(resp_len);
1550 int total_bytes = recv(fd, (char*)resp, (int)resp_len, 0);
1551 if (total_bytes == (int)resp_len && resp_len >= 10) {
1552 /* 检查是否成功 */
1553 int success = resp[9] != 0;
1554 if (success) result = 0;
1555 else {
1556 /* 如果响应中有新的 leader 信息，更新 */
1557 uint32_t lid_len = read_u32(resp + 10);
1558 if (lid_len > 0 && lid_len < RAFT_NODE_ID_LEN && 14 + lid_len <= resp_len) {
1559 memcpy(r->leader_peer.id, resp + 14, lid_len);
1560 r->leader_peer.id[lid_len] = '\0';
1561 uint32_t host_off = (uint32_t)(14 + lid_len);
1562 if (host_off + 4 <= resp_len) {
1563 uint32_t host_len = read_u32(resp + host_off);
1564 if (host_len > 0 && host_len < 64 && host_off + 4 + host_len + 4 <= resp_len) {
1565 memcpy(r->leader_peer.host, resp + host_off + 4, host_len);
1566 r->leader_peer.host[host_len] = '\0';
1567 r->leader_peer.port = (int)read_u32(resp + host_off + 4 + host_len);
1568 r->leader_known = 1;
1569 }
1570 }
1571 }
1572 }
1573 }
1574 kv_free(resp);
1575 }
1576 close_socket(fd);
1577 MUTEX_UNLOCK(&r->mutex);
1578 return result;
1579 }
1580 MUTEX_UNLOCK(&r->mutex);
1581 return -1;
1582 }
1583
1584 MUTEX_UNLOCK(&r->mutex);
1585
1586 /* 我们是 Leader，直接在本地处理 */
1587 /* 通过 propose RPC 自处理（简化：直接通过内部 API 处理） */
1588 MUTEX_LOCK(&r->mutex);
1589 if (r->log_count >= r->log_capacity) {
1590 r->log_capacity *= 2;
1591 void* tmp = kv_realloc(r->log, r->log_capacity * sizeof(raft_log_entry_t));
1592 if (!tmp) { MUTEX_UNLOCK(&r->mutex); return -1; }
1593 r->log = (raft_log_entry_t*)tmp;
1594 }
1595
1596 uint64_t new_idx = r->log_count > 0 ? r->log[r->log_count - 1].index + 1 : 1;
1597 raft_log_entry_t* ne = &r->log[r->log_count++];
1598 ne->term = r->current_term;
1599 ne->index = new_idx;
1600 ne->type = type;
1601 ne->key = kv_malloc(key_len);
1602 memcpy(ne->key, key, key_len);
1603 ne->key_len = key_len;
1604 ne->value = value_len > 0 ? kv_malloc(value_len) : NULL;
1605 if (ne->value) memcpy(ne->value, value, value_len);
1606 ne->value_len = value_len;
1607
1608 r->log_size_bytes += raft_log_entry_size(ne);
1609
1610 r->match_index[raft_self_index(r)] = new_idx;
1611 MUTEX_UNLOCK(&r->mutex);
1612 raft_log_save(r);
1613
1614 /* 复制到 followers */
1615 raft_leader_send_heartbeat(r);
1616
1617 return 0;
1618}

1719 uint64_t last_included_term = read_u64(data + off); off += 8;
1720 uint64_t snap_data_len = read_u64(data + off); off += 8;

1724
1725 if (term < r->current_term) {
1726 MUTEX_UNLOCK(&r->mutex);
1727 rpc_send_snapshot_response(fd, r->current_term, 0);
1728 return;
1729 }
1730
1731 r->last_heartbeat_ms = thread_time_ms();
1732
1733 /* 如果快照比当前日志新，则接受 */
1734 if (last_included_index > r->commit_index) {

1737 snprintf(snap_path, sizeof(snap_path), "%s/snapshot.dat", r->cfg.data_dir);

1757

1948/* 向 Follower 发送 InstallSnapshot（支持分块） */
1949static void raft_leader_send_snapshot(raft_t* r, int peer_index) {
1950 if (!r->cfg.snapshot_fn) return;
1951
1952 /* 先创建快照（如果尚未创建） */
1953 if (r->snapshot_index == 0) {
1954 raft_snapshot_create(r);
1955 }
1956
1957 if (r->snapshot_index == 0) return;
1958
1959 /* 读取快照文件 */
1960 FILE* f = fopen(r->snapshot_path, "rb");
1961 if (!f) return;
1962
1963 fseek(f, 0, SEEK_END);
1964 long file_size = ftell(f);
1965 fseek(f, 16, SEEK_SET); /* 跳过头部 (index + term) */
1966
1967 size_t data_len = (size_t)(file_size - 16);
1968 if (data_len == 0) { fclose(f); return; }
1969
1970 /* 小文件直接发送，大文件分块发送 */
1971 if (data_len <= RAFT_SNAPSHOT_CHUNK_SIZE) {
1972 uint8_t* snap_data = kv_malloc(data_len);
1973 if (!snap_data) { fclose(f); return; }
1974 fread(snap_data, 1, data_len, f);
1975 fclose(f);
1976
1977 SOCKET fd = raft_connect_peer(&r->cfg.peers[peer_index]);
1978 if (fd == INVALID_SOCKET) { kv_free(snap_data); return; }
1979
1980 rpc_send_snapshot_request(fd, r->current_term, r->cfg.node_id,
1981 r->snapshot_index, r->snapshot_term,
1982 snap_data, data_len);
1983
1984 uint32_t resp_len = 0;
1985 int ret = recv(fd, (char*)&resp_len, 4, 0);
1986 if (ret == 4) {
1987 uint8_t resp_buf[64];
1988 int total = recv(fd, (char*)resp_buf, (int)resp_len, 0);
1989 if (total == (int)resp_len) {
1990 raft_handle_snapshot_response(r, peer_index, resp_buf, resp_len);
1991 }
1992 }
1993 close_socket(fd);
1994 kv_free(snap_data);
1995 } else {
1996 /* 分块传输 */
1997 uint8_t* chunk_buf = kv_malloc(RAFT_SNAPSHOT_CHUNK_SIZE);
1998 if (!chunk_buf) { fclose(f); return; }
1999
2000 uint64_t offset = 0;
2001 while (offset < data_len) {
2002 size_t chunk_size = RAFT_SNAPSHOT_CHUNK_SIZE;
2003 if (offset + chunk_size > data_len) {
2004 chunk_size = data_len - offset;
2005 }
2006
2007 fseek(f, 16 + (long)offset, SEEK_SET);
2008 fread(chunk_buf, 1, chunk_size, f);
2009
2010 SOCKET fd = raft_connect_peer(&r->cfg.peers[peer_index]);
2011 if (fd == INVALID_SOCKET) { kv_free(chunk_buf); fclose(f); return; }
2012
2013 int is_last = (offset + chunk_size >= data_len) ? 1 : 0;
2014
2015 /* 发送分块 */
2016 uint8_t hdr[64];
2017 size_t hdr_off = 0;
2018 hdr[hdr_off++] = RAFT_RPC_SNAPSHOT_CHUNK;
2019 write_u64(hdr + hdr_off, r->current_term); hdr_off += 8;
2020 uint32_t lid_len = (uint32_t)strlen(r->cfg.node_id);
2021 write_u32(hdr + hdr_off, lid_len); hdr_off += 4;
2022 memcpy(hdr + hdr_off, r->cfg.node_id, lid_len); hdr_off += lid_len;
2023 write_u64(hdr + hdr_off, r->snapshot_index); hdr_off += 8;
2024 write_u64(hdr + hdr_off, r->snapshot_term); hdr_off += 8;
2025 write_u64(hdr + hdr_off, (uint64_t)data_len); hdr_off += 8; /* total_size */
2026 write_u64(hdr + hdr_off, offset); hdr_off += 8;
2027 write_u64(hdr + hdr_off, (uint64_t)chunk_size); hdr_off += 8;
2028 hdr[hdr_off++] = (uint8_t)is_last;
2029
2030 uint32_t total = (uint32_t)(hdr_off + chunk_size);
2031 send(fd, (const char*)&total, 4, 0);
2032 send(fd, (const char*)hdr, (int)hdr_off, 0);
2033 send(fd, (const char*)chunk_buf, (int)chunk_size, 0);
2034
2035 /* 等待 ACK */
2036 uint32_t resp_len = 0;
2037 int ret = recv(fd, (char*)&resp_len, 4, 0);
2038 if (ret == 4) {
2039 uint8_t resp_buf[64];
2040 int total_bytes = recv(fd, (char*)resp_buf, (int)resp_len, 0);
2041 if (total_bytes == (int)resp_len && resp_len >= 10) {
2042 if (resp_buf[9] == 0) {
2043 /* chunk rejected */
2044 close_socket(fd);
2045 kv_free(chunk_buf);
2046 fclose(f);
2047 return;
2048 }
2049 }
2050 }
2051 close_socket(fd);
2052 offset += chunk_size;
2053 }
2054
2055 /* 所有分块发送完毕，更新 match_index */
2056 if (offset >= data_len) {
2057 /* 手动更新 match_index（快照传输完成） */
2058 MUTEX_LOCK(&r->mutex);
2059 if (r->snapshot_index > r->match_index[peer_index]) {
2060 r->match_index[peer_index] = r->snapshot_index;
2061 r->next_index[peer_index] = r->snapshot_index + 1;
2062 }
2063 MUTEX_UNLOCK(&r->mutex);
2064 }
2065
2066 kv_free(chunk_buf);
2067 fclose(f);
2068 }
2069}
