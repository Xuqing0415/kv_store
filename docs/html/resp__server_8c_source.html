74/* 读取一行直到 \r\n，返回行内容（不含 \r\n），需要调用者释放 */
75static char* resp_read_line(SOCKET fd, char* buf, size_t* buf_pos, size_t* buf_len) {
76 (void)fd;
77 /* 在缓冲区中查找 \r\n */
78 for (size_t i = *buf_pos; i + 1 < *buf_len; i++) {
79 if (buf[i] == '\r' && buf[i + 1] == '\n') {
80 size_t line_len = i - *buf_pos;
81 char* line = kv_malloc(line_len + 1);
82 if (!line) return NULL;
83 memcpy(line, buf + *buf_pos, line_len);
84 line[line_len] = '\0';
85 *buf_pos = i + 2;
86 return line;
87 }
88 }
89 return NULL; /* 行不完整，需要更多数据 */
90}

101/* 解析 RESP 协议 */
102static resp_value_t* resp_parse(SOCKET fd, char* buf, size_t* buf_pos, size_t* buf_len) {
103 if (*buf_pos >= *buf_len) {
104 if (resp_fill_buffer(fd, buf, buf_len, RESP_MAX_CMD_LEN) != 0) return NULL;
105 }
106
107 if (*buf_pos >= *buf_len) return NULL;
108
109 char type = buf[*buf_pos];
110 (*buf_pos)++;
111
112 char* line = resp_read_line(fd, buf, buf_pos, buf_len);
113 if (!line) return NULL;
114
115 resp_value_t* v = NULL;
116
117 switch (type) {
118 case '+': { /* Simple String */
119 v = resp_value_new(RESP_STRING);
120 if (v) {
121 v->str = line;
122 v->len = strlen(line);
123 } else {
124 kv_free(line);
125 }
126 return v;
127 }
128 case '-': { /* Error */
129 v = resp_value_new(RESP_ERROR);
130 if (v) {
131 v->str = line;
132 v->len = strlen(line);
133 } else {
134 kv_free(line);
135 }
136 return v;
137 }
138 case ':': { /* Integer */
139 v = resp_value_new(RESP_INTEGER);
140 if (v) {
141 v->integer = strtoll(line, NULL, 10);
142 }
143 kv_free(line);
144 return v;
145 }
146 case '$': { /* Bulk String */
147 long long bulk_len = strtoll(line, NULL, 10);
148 kv_free(line);
149
150 if (bulk_len < 0) {
151 /* Null bulk string */
152 v = resp_value_new(RESP_BULK);
153 if (v) {
154 v->str = NULL;
155 v->len = 0;
156 }
157 return v;
158 }
159
160 if (bulk_len > RESP_MAX_CMD_LEN) return NULL;
161
162 v = resp_value_new(RESP_BULK);
163 if (!v) return NULL;
164
165 /* 确保缓冲区有足够数据: bulk_len + \r\n */
166 while (*buf_len - *buf_pos < (size_t)bulk_len + 2) {
167 if (resp_fill_buffer(fd, buf, buf_len, RESP_MAX_CMD_LEN) != 0) {
168 resp_value_free(v);
169 return NULL;
170 }
171 }
172
173 v->str = kv_malloc((size_t)bulk_len + 1);
174 if (!v->str) {
175 resp_value_free(v);
176 return NULL;
177 }
178 memcpy(v->str, buf + *buf_pos, (size_t)bulk_len);
179 v->str[bulk_len] = '\0';
180 v->len = (size_t)bulk_len;
181 *buf_pos += (size_t)bulk_len;
182
183 /* 读取 \r\n */
184 line = resp_read_line(fd, buf, buf_pos, buf_len);
185 if (line) kv_free(line);
186
187 return v;
188 }
189 case '*': { /* Array */
190 long long arr_count = strtoll(line, NULL, 10);
191 kv_free(line);
192
193 if (arr_count < 0 || arr_count > 256) return NULL;
194
195 v = resp_value_new(RESP_ARRAY);
196 if (!v) return NULL;
197
198 v->count = (size_t)arr_count;
199 if (v->count == 0) return v;
200
201 v->elements = kv_calloc(v->count, sizeof(resp_value_t*));
202 if (!v->elements) {
203 resp_value_free(v);
204 return NULL;
205 }
206
207 for (size_t i = 0; i < v->count; i++) {
208 v->elements[i] = resp_parse(fd, buf, buf_pos, buf_len);
209 if (!v->elements[i]) {
210 resp_value_free(v);
211 return NULL;
212 }
213 }
214
215 return v;
216 }
217 default:
218 kv_free(line);
219 return NULL;
220 }
221}

500/* 处理客户端请求 */
501static void resp_process_client(resp_server_t* server, resp_client_t* client) {
502 /* 读取更多数据 */
503 if (client->buf_len < RESP_IO_BUF_SIZE) {
504 int n = recv(client->fd, client->read_buf + client->buf_len,
505 (int)(RESP_IO_BUF_SIZE - client->buf_len), 0);
506 if (n <= 0) {
507 client->active = 0;
508 return;
509 }
510 client->buf_len += n;
511 }
512
513 /* 尝试解析命令 */
514 while (client->buf_pos < client->buf_len) {
515 size_t saved_pos = client->buf_pos;
516 size_t saved_len = client->buf_len;
517
518 resp_value_t* cmd = resp_parse(client->fd, client->read_buf,
519 &client->buf_pos, &client->buf_len);
520 if (!cmd) {
521 /* 解析失败或数据不完整 */
522 client->buf_pos = saved_pos;
523 client->buf_len = saved_len;
524 /* 紧凑缓冲区 */
525 if (client->buf_pos > 0 && client->buf_len > client->buf_pos) {
526 memmove(client->read_buf, client->read_buf + client->buf_pos,
527 client->buf_len - client->buf_pos);
528 client->buf_len -= client->buf_pos;
529 client->buf_pos = 0;
530 } else if (client->buf_pos >= client->buf_len) {
531 client->buf_pos = 0;
532 client->buf_len = 0;
533 }
534 return;
535 }
536
537 /* 紧凑缓冲区 */
538 if (client->buf_pos > 0 && client->buf_len > client->buf_pos) {
539 memmove(client->read_buf, client->read_buf + client->buf_pos,
540 client->buf_len - client->buf_pos);
541 client->buf_len -= client->buf_pos;
542 client->buf_pos = 0;
543 } else if (client->buf_pos >= client->buf_len) {
544 client->buf_pos = 0;
545 client->buf_len = 0;
546 }
547
548 if (cmd->type != RESP_ARRAY || cmd->count == 0) {
549 resp_value_free(cmd);
550 continue;
551 }
552
553 /* 获取命令名 */
554 if (cmd->elements[0]->type != RESP_BULK || !cmd->elements[0]->str) {
555 resp_value_free(cmd);
556 continue;
557 }
558
559 char* cmd_name = cmd->elements[0]->str;
560 str_tolower(cmd_name);
561
562 resp_reply_t* reply = resp_reply_new();
563 if (!reply) {
564 resp_value_free(cmd);
565 continue;
566 }
567
568 /* 路由命令 */
569 if (strcmp(cmd_name, "ping") == 0) {
570 cmd_ping(reply, cmd);
571 } else if (strcmp(cmd_name, "set") == 0) {
572 cmd_set(server->db, reply, cmd);
573 } else if (strcmp(cmd_name, "get") == 0) {
574 cmd_get(server->db, reply, cmd);
575 } else if (strcmp(cmd_name, "del") == 0) {
576 cmd_del(server->db, reply, cmd);
577 } else if (strcmp(cmd_name, "exists") == 0) {
578 cmd_exists(server->db, reply, cmd);
579 } else if (strcmp(cmd_name, "dbsize") == 0) {
580 cmd_dbsize(server->db, reply, cmd);
581 } else if (strcmp(cmd_name, "flushdb") == 0) {
582 cmd_flushdb(server->db, reply, cmd);
583 } else if (strcmp(cmd_name, "info") == 0) {
584 cmd_info(server->db, reply, cmd);
585 } else if (strcmp(cmd_name, "quit") == 0) {
586 cmd_quit(reply, cmd);
587 } else if (strcmp(cmd_name, "save") == 0) {
588 cmd_save(server->db, reply, cmd);
589 } else if (strcmp(cmd_name, "bgsave") == 0) {
590 cmd_bgsave(server->db, reply, cmd);
591 } else if (strcmp(cmd_name, "command") == 0) {
592 resp_make_ok(reply); /* 兼容 redis-cli */
593 } else {
594 char err[128];
595 snprintf(err, sizeof(err), "unknown command '%s'", cmd_name);
596 resp_make_error(reply, err);
597 }
598
599 /* 发送响应 */
600 if (reply->len > 0) {
601 send(client->fd, reply->buf, (int)reply->len, 0);
602 }
603
604 resp_reply_free(reply);
605 resp_value_free(cmd);
606
607 /* QUIT 命令关闭连接 */
608 if (strcmp(cmd_name, "quit") == 0) {
609 client->active = 0;
610 return;
611 }
612 }
613}

619int resp_server_start(resp_server_t** out_server, const char* host, int port, kv_store_t* db) {
620 if (!out_server || !db) return -1;
621
622 resp_server_t* server = kv_malloc(sizeof(resp_server_t));
623 if (!server) return -1;
624 memset(server, 0, sizeof(resp_server_t));
625
626 server->db = db;
627 server->running = 0;
628 server->host = kv_strdup(host ? host : "127.0.0.1");
629 server->port = port;
630
631#ifdef _WIN32
632 WSADATA wsa_data;
633 if (WSAStartup(MAKEWORD(2, 2), &wsa_data) != 0) {
634 kv_free(server->host);
635 kv_free(server);
636 return -1;
637 }
638 server->wsock_initialized = 1;
639#endif
640
641 server->listen_fd = socket(AF_INET, SOCK_STREAM, 0);
642 if (server->listen_fd == INVALID_SOCKET) {
643 printf("[RESP] Failed to create socket\n");
644 resp_server_free(server);
645 return -1;
646 }
647
648 /* 设置 SO_REUSEADDR */
649 int opt = 1;
650 setsockopt(server->listen_fd, SOL_SOCKET, SO_REUSEADDR,
651 (const char*)&opt, sizeof(opt));
652
653 /* 设置 TCP_NODELAY */
654 setsockopt(server->listen_fd, IPPROTO_TCP, TCP_NODELAY,
655 (const char*)&opt, sizeof(opt));
656
657 struct sockaddr_in addr;
658 memset(&addr, 0, sizeof(addr));
659 addr.sin_family = AF_INET;
660 addr.sin_port = htons((unsigned short)port);
661 addr.sin_addr.s_addr = inet_addr(server->host);
662
663 if (bind(server->listen_fd, (struct sockaddr*)&addr, sizeof(addr)) == SOCKET_ERROR) {
664 printf("[RESP] Failed to bind %s:%d\n", server->host, server->port);
665 resp_server_free(server);
666 return -1;
667 }
668
669 if (listen(server->listen_fd, SOMAXCONN) == SOCKET_ERROR) {
670 printf("[RESP] Failed to listen\n");
671 resp_server_free(server);
672 return -1;
673 }
674
675 /* 初始化客户端 */
676 for (int i = 0; i < RESP_MAX_CLIENTS; i++) {
677 server->clients[i].fd = INVALID_SOCKET;
678 server->clients[i].active = 0;

686void resp_server_run(resp_server_t* server) {
687 if (!server) return;
688
689 server->running = 1;
690 printf("[RESP] Server started, accepting connections...\n");
691
692 while (server->running) {
693 fd_set read_fds;
694 FD_ZERO(&read_fds);
695 FD_SET(server->listen_fd, &read_fds);
696
697 SOCKET max_fd = server->listen_fd;
698
699 for (int i = 0; i < RESP_MAX_CLIENTS; i++) {
700 if (server->clients[i].active) {
701 FD_SET(server->clients[i].fd, &read_fds);
702 if (server->clients[i].fd > max_fd) {
703 max_fd = server->clients[i].fd;
704 }
705 }
706 }
707
708 struct timeval tv;
709 tv.tv_sec = 1;
710 tv.tv_usec = 0;
711
712 int ret = select((int)max_fd + 1, &read_fds, NULL, NULL, &tv);
713 if (ret < 0) {
714 if (
715#ifdef _WIN32
716 WSAGetLastError() == WSAEINTR
717#else
718 errno == EINTR
719#endif
720 ) {
721 continue;
722 }
723 break;
724 }
725
726 if (ret == 0) continue;
727
728 /* 接受新连接 */
729 if (FD_ISSET(server->listen_fd, &read_fds)) {
730 struct sockaddr_in client_addr;
731 socklen_t addr_len = sizeof(client_addr);
732 SOCKET client_fd = accept(server->listen_fd,
733 (struct sockaddr*)&client_addr, &addr_len);
734 if (client_fd != INVALID_SOCKET) {
735 /* 设置 TCP_NODELAY */
736 int opt = 1;
737 setsockopt(client_fd, IPPROTO_TCP, TCP_NODELAY,
738 (const char*)&opt, sizeof(opt));
739
740 int added = 0;
741 for (int i = 0; i < RESP_MAX_CLIENTS; i++) {
742 if (!server->clients[i].active) {
743 server->clients[i].fd = client_fd;
744 server->clients[i].buf_pos = 0;
745 server->clients[i].buf_len = 0;
746 server->clients[i].active = 1;
747 added = 1;
748 printf("[RESP] Client connected: %s\n",
749 inet_ntoa(client_addr.sin_addr));
750 break;
751 }
752 }
753 if (!added) {
754 printf("[RESP] Max clients reached, rejecting connection\n");
755 close_socket(client_fd);
756 }
757 }
758 }
759
760 /* 处理客户端数据 */
761 for (int i = 0; i < RESP_MAX_CLIENTS; i++) {
762 if (server->clients[i].active && FD_ISSET(server->clients[i].fd, &read_fds)) {
763 resp_process_client(server, &server->clients[i]);
764 if (!server->clients[i].active) {
765 printf("[RESP] Client disconnected\n");
766 close_socket(server->clients[i].fd);
767 server->clients[i].fd = INVALID_SOCKET;
768 }
769 }
770 }
771 }
772}
