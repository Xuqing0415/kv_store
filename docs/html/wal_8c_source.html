50/* 扫描目录，找到所有 WAL 文件的最大序列号，返回 -1 表示没有文件 */
51static int wal_mgr_scan_dir(const char* dir_path) {
52 int max_seq = -1;
53
54#ifdef _WIN32
55 char pattern[512];
56 snprintf(pattern, sizeof(pattern), "%s/" WAL_FILE_PREFIX "*" WAL_FILE_SUFFIX, dir_path);
57
58 WIN32_FIND_DATAA fd;
59 HANDLE hFind = FindFirstFileA(pattern, &fd);
60 if (hFind == INVALID_HANDLE_VALUE) return -1;
61
62 do {
63 /* 从文件名中提取序列号：wal_XXXXXX.log */
64 const char* name = fd.cFileName;
65 const char* prefix_pos = strstr(name, WAL_FILE_PREFIX);
66 if (!prefix_pos) continue;
67 prefix_pos += strlen(WAL_FILE_PREFIX);
68
69 int seq = 0;
70 int digits = 0;
71 while (*prefix_pos >= '0' && *prefix_pos <= '9' && digits < WAL_SEQ_DIGITS) {
72 seq = seq * 10 + (*prefix_pos - '0');
73 prefix_pos++;
74 digits++;
75 }
76 if (digits == WAL_SEQ_DIGITS && seq > max_seq) {
77 max_seq = seq;
78 }
79 } while (FindNextFileA(hFind, &fd));
80
81 FindClose(hFind);
82#else
83 DIR* dir = opendir(dir_path);
84 if (!dir) return -1;
85
86 struct dirent* entry;
87 while ((entry = readdir(dir)) != NULL) {
88 const char* name = entry->d_name;
89 if (strncmp(name, WAL_FILE_PREFIX, strlen(WAL_FILE_PREFIX)) != 0) continue;
90
91 const char* p = name + strlen(WAL_FILE_PREFIX);
92 /* 检查后缀 */
93 const char* suffix = strstr(p, WAL_FILE_SUFFIX);
94 if (!suffix) continue;
95
96 int seq = 0;
97 int digits = 0;
98 while (p < suffix && *p >= '0' && *p <= '9' && digits < WAL_SEQ_DIGITS) {
99 seq = seq * 10 + (*p - '0');
100 p++;
101 digits++;
102 }
103 if (digits == WAL_SEQ_DIGITS && seq > max_seq) {
104 max_seq = seq;
105 }
106 }
107 closedir(dir);
108#endif
109
110 return max_seq;

345int wal_write(wal_t* w, wal_record_type_t type, const char* key, size_t klen, const char* value, size_t vlen) {
346 if (!w || !w->file || !key || klen == 0) return -1;
347
348 size_t total_size = WAL_HEADER_SIZE + 1 + 4 + klen + 4 + vlen + WAL_CRC_SIZE;
349 uint8_t* buf = kv_malloc(total_size);
350 if (!buf) return -1;
351
352 uint8_t* ptr = buf;
353
354 ptr += encode_fixed32((uint32_t)(total_size - WAL_HEADER_SIZE), ptr);
355
356 *ptr++ = (uint8_t)type;
357
358 ptr += encode_fixed32((uint32_t)klen, ptr);
359 memcpy(ptr, key, klen);
360 ptr += klen;
361
362 ptr += encode_fixed32((uint32_t)vlen, ptr);
363 if (value) {
364 memcpy(ptr, value, vlen);
365 }
366 ptr += vlen;
367
368 uint32_t crc = crc32(buf + WAL_HEADER_SIZE, total_size - WAL_HEADER_SIZE - WAL_CRC_SIZE);
369 ptr += encode_fixed32(crc, ptr);
370
371 size_t written = fwrite(buf, 1, total_size, w->file);
372 kv_free(buf);
373
374 if (written != total_size) {
375 return -1;
376 }
377
378 w->bytes_since_sync += total_size;
379 if (w->bytes_since_sync >= w->sync_interval) {
380 wal_sync(w);
381 }
382
383 return 0;
384}

396int wal_replay(wal_t* w, int (*callback)(wal_record_t* record, void* arg), void* arg) {
397 if (!w || !w->file || !callback) return -1;
398
399 if (fseek(w->file, 0, SEEK_SET) != 0) return -1;
400
401 while (1) {
402 uint32_t record_len;
403 if (fread(&record_len, 4, 1, w->file) != 1) {
404 if (feof(w->file)) break;
405 return -1;
406 }
407
408 size_t total_len = record_len + WAL_HEADER_SIZE;
409 uint8_t* buf = kv_malloc(total_len);
410 if (!buf) return -1;
411
412 encode_fixed32(record_len, buf);
413
414 if (fread(buf + WAL_HEADER_SIZE, 1, record_len, w->file) != record_len) {
415 kv_free(buf);
416 if (feof(w->file)) break;
417 return -1;
418 }
419
420 uint32_t stored_crc;
421 decode_fixed32(buf + total_len - WAL_CRC_SIZE, &stored_crc);
422
423 uint32_t computed_crc = crc32(buf + WAL_HEADER_SIZE, record_len - WAL_CRC_SIZE);
424 if (stored_crc != computed_crc) {
425 kv_free(buf);
426 break;
427 }
428
429 wal_record_t record;
430 uint8_t* ptr = buf + WAL_HEADER_SIZE;
431
432 record.type = (wal_record_type_t)*ptr++;
433
434 uint32_t klen;
435 ptr += decode_fixed32(ptr, &klen);
436 record.key = kv_malloc(klen);
437 memcpy(record.key, ptr, klen);
438 record.key_len = klen;
439 ptr += klen;
440
441 uint32_t vlen;
442 ptr += decode_fixed32(ptr, &vlen);
443 if (vlen > 0) {
444 record.value = kv_malloc(vlen);
445 memcpy(record.value, ptr, vlen);
446 } else {
447 record.value = NULL;
448 }
449 record.value_len = vlen;
450
451 int ret = callback(&record, arg);
452
453 kv_free(record.key);
454 kv_free(record.value);
455 kv_free(buf);
456
457 if (ret != 0) return ret;
458 }
459
460 return 0;
461}
