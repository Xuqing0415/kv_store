63static int skiplist_key_compare(const char* a, size_t a_len, const char* b, size_t b_len) {
64 if (a_len == 0 && b_len == 0) return 0;
65 if (a_len == 0) return -1;
66 if (b_len == 0) return 1;
67 size_t min_len = a_len < b_len ? a_len : b_len;
68 int cmp = memcmp(a, b, min_len);
69 if (cmp != 0) return cmp;
70 if (a_len < b_len) return -1;
71 if (a_len > b_len) return 1;
72 return 0;
73}

128 if (!sl || !key || klen == 0) return -1;
129

131 skiplist_node_t* x = sl->header;
132
133 for (int i = sl->level - 1; i >= 0; i--) {
134 while (x->forward[i] && skiplist_key_compare(x->forward[i]->key, x->forward[i]->key_len, key, klen) < 0) {

138 }
139
140 x = x->forward[0];
141

143 sl->memory_usage -= x->value_len;
144 kv_free(x->value);
145 if (value && vlen > 0) {

153 x->deleted = 0;
154 sl->memory_usage += vlen;
155 return 0;
156 }
157

159 if (level > sl->level) {
160 for (int i = sl->level; i < level; i++) {
161 update[i] = sl->header;

164 }

167 if (!x) return -1;

172 }
