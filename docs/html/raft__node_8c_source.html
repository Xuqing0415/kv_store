21/* ================================================================
22 * Raft 状态机回调：将 committed 日志应用到 kv_store
23 * ================================================================ */
24static int raft_apply_to_kv(void* state, raft_entry_t* entry) {
25 kv_store_t* db = (kv_store_t*)state;
26 if (!db || !entry) return -1;
27
28 if (entry->type == 0) { /* PUT */
29 return kv_put_internal(db, entry->key, entry->key_len, entry->value, entry->value_len);
30 } else if (entry->type == 1) { /* DELETE */
31 return kv_delete_internal(db, entry->key, entry->key_len);
32 }
33 return -1;
34}

574 /* 解析命令行参数 */
575 for (int i = 1; i < argc; i++) {

579 raft_port = atoi(argv[++i]);

582 } else if (strcmp(argv[i], "--metrics-port") == 0 && i + 1 < argc) {
583 metrics_port = atoi(argv[++i]);

589 return 1;
590 }
591 char spec[256];
592 strncpy(spec, argv[++i], sizeof(spec) - 1);
593 spec[sizeof(spec) - 1] = '\0';
594
595 /* 解析 node_id:host:port */

598 char* port_part = strtok(NULL, ":");

607 peers[num_peers].port = atoi(port_part);

622 return 1;
623 }
624

692 printf("[INFO] No MANIFEST found, restoring from snapshot (index=%llu)...\n",
693 (unsigned long long)snap_info.last_included_index);
694 if (raft_snapshot_restore(g_raft, snap_info.file_path) != 0) {
695 fprintf(stderr, "WARNING: Failed to restore from snapshot %s\n",
696 snap_info.file_path);
697 }
698 }
699 }
700 }

702 /* 崩溃恢复：重放已提交但未应用的 Raft 日志到 KV 引擎 */
703 {

705 raft_role_t role;
706 size_t log_count;
707 raft_status(g_raft, &term, &role, &commit_idx, &last_applied, &log_count, NULL);
708 printf("[INFO] Raft state: term=%llu, role=%s, commit=%llu, applied=%llu, log_count=%zu\n",
709 (unsigned long long)term, raft_role_str(role),

726 printf("[INFO] Waiting for leader election...\n");

736 break;
737 }
738
739 const raft_peer_t* leader = raft_get_leader_peer(g_raft);
740 if (leader && leader->port > 0) {

742 leader->id, leader->host, leader->port,
743 raft_role_str(role));

748#else
749 usleep(100000);
750#endif

757 } else {

762 g_resp_started = 1;
763 }
764#endif

766 }
767
768 /* 启动 Metrics 服务器（后台线程） */
769 if (metrics_port > 0) {
770 if (metrics_server_start(&g_metrics, "0.0.0.0", metrics_port, g_db, g_raft) != 0) {
771 fprintf(stderr, "WARNING: Failed to start metrics server on port %d\n", metrics_port);
772 } else {
773#ifdef _WIN32
774 g_metrics_thread = CreateThread(NULL, 0, metrics_thread_func, g_metrics, 0, NULL);
775#else
776 if (pthread_create(&g_metrics_thread, NULL, metrics_thread_func, g_metrics) == 0) {
777 g_metrics_started = 1;
778 }
779#endif
780 printf("[INFO] Metrics server listening on 0.0.0.0:%d\n", metrics_port);
781 }
782 }
783
784 /* 注册信号处理 */
785#ifdef _WIN32
786 SetConsoleCtrlHandler(signal_handler, TRUE);
787#else
788 signal(SIGINT, signal_handler);
789 signal(SIGTERM, signal_handler);
790#endif
791
792 printf("\n========================================\n");
793 printf(" Cluster node ready!\n");
794 printf(" RESP: redis-cli -h 127.0.0.1 -p %d\n", resp_port);
795 if (metrics_port > 0) {
796 printf(" Metrics: http://127.0.0.1:%d/metrics\n", metrics_port);
797 }
798 printf(" Press Ctrl+C to stop.\n");
799 printf("========================================\n\n");
800
801 /* 主线程等待（直到信号停止） */
802 while (g_raft) {
803 raft_status(g_raft, NULL, NULL, NULL, NULL, NULL, NULL);
804#ifdef _WIN32
805 Sleep(1000);
806#else
807 sleep(1);
808#endif
809 }
810
811 /* 清理 */
812 printf("[INFO] Cleaning up...\n");
813
814 if (g_metrics) metrics_server_stop(g_metrics);
815 if (g_resp) resp_server_stop(g_resp);
816 if (g_raft) { raft_stop(g_raft); raft_join(g_raft); }
817
818#ifdef _WIN32
819 if (g_metrics_thread) {
820 WaitForSingleObject(g_metrics_thread, 5000);
821 CloseHandle(g_metrics_thread);
822 }
823 if (g_resp_thread) {
824 WaitForSingleObject(g_resp_thread, 5000);
825 CloseHandle(g_resp_thread);
826 }
827#else
828 if (g_metrics_started) {
829 pthread_join(g_metrics_thread, NULL);
830 }
831 if (g_resp_started) {
832 pthread_join(g_resp_thread, NULL);
833 }
834#endif
835
836 if (g_metrics) metrics_server_free(g_metrics);
837 if (g_resp) resp_server_free(g_resp);
838 if (g_raft) raft_destroy(g_raft);
839 if (g_db) kv_close(g_db);
840
841 printf("[INFO] Node stopped.\n");
842 return 0;
