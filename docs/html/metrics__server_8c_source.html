50/* 尝试从 socket 读取 HTTP 请求行，直到读到 \r\n\r\n 或缓冲区满 */
51static int http_read_request(SOCKET fd, char* buf, size_t* buf_len, size_t buf_capacity) {
52 *buf_len = 0;
53 while (*buf_len < buf_capacity - 1) {
54 int n = recv(fd, buf + *buf_len, (int)(buf_capacity - 1 - *buf_len), 0);
55 if (n <= 0) {
56 return -1; /* 连接关闭或错误 */
57 }
58 *buf_len += n;
59 buf[*buf_len] = '\0';
60
61 /* 检查是否收到完整的请求头（以 \r\n\r\n 结束） */
62 if (strstr(buf, "\r\n\r\n") != NULL) {
63 return 0;
64 }
65 }
66 return -1; /* 请求太大 */
67}

69/* 解析 HTTP 请求行，提取方法（GET/POST）和路径 */
70static int http_parse_request(const char* buf, size_t buf_len,
71 char* method, size_t method_cap,
72 char* path, size_t path_cap) {
73 (void)buf_len;
74 /* 解析第一行: METHOD SP PATH SP HTTP/1.x\r\n */
75 const char* line_end = strstr(buf, "\r\n");
76 if (!line_end) return -1;
77
78 size_t line_len = (size_t)(line_end - buf);
79 if (line_len > 4096) return -1; /* 安全限制 */
80
81 char line[4096];
82 size_t copy_len = line_len < sizeof(line) - 1 ? line_len : sizeof(line) - 1;
83 memcpy(line, buf, copy_len);
84 line[copy_len] = '\0';
85
86 /* 提取 method */
87 const char* sp1 = strchr(line, ' ');
88 if (!sp1) return -1;
89 size_t method_len = (size_t)(sp1 - line);
90 if (method_len >= method_cap) return -1;
91 memcpy(method, line, method_len);
92 method[method_len] = '\0';
93
94 /* 提取 path */
95 const char* sp2 = strchr(sp1 + 1, ' ');
96 if (sp2) {
97 size_t path_len = (size_t)(sp2 - (sp1 + 1));
98 if (path_len >= path_cap) return -1;
99 memcpy(path, sp1 + 1, path_len);
100 path[path_len] = '\0';
101 } else {
102 /* 没有第二段空格，path 就是到行尾 */
103 size_t path_len = line_len - (size_t)(sp1 + 1 - line);
104 if (path_len >= path_cap) return -1;
105 memcpy(path, sp1 + 1, path_len);
106 path[path_len] = '\0';
107 }
108
109 return 0;
110}

249static void metrics_handle_connection(metrics_server_t* server, SOCKET client_fd) {
250 char buf[HTTP_BUF_SIZE];
251 size_t buf_len = 0;
252
253 /* 读取 HTTP 请求 */
254 if (http_read_request(client_fd, buf, &buf_len, sizeof(buf)) != 0) {
255 close_socket(client_fd);
256 return;
257 }
258
259 char method[16] = {0};
260 char path[256] = {0};
261
262 if (http_parse_request(buf, buf_len, method, sizeof(method), path, sizeof(path)) != 0) {
263 close_socket(client_fd);
264 return;
265 }
266
267 char resp[METRICS_BUF_SIZE];
268 size_t resp_len = 0;
269
270 /* 只处理 GET /metrics */
271 if (strcmp(method, "GET") == 0 && strcmp(path, "/metrics") == 0) {
272 if (metrics_build_response(server, server->start_time, resp, sizeof(resp), &resp_len) != 0) {
273 close_socket(client_fd);
274 return;
275 }
276 } else {
277 if (metrics_build_404(resp, sizeof(resp), &resp_len) != 0) {
278 close_socket(client_fd);
279 return;
280 }
281 }
282
283 /* 发送响应 */
284 send(client_fd, resp, (int)resp_len, 0);
285 close_socket(client_fd);
286}

356void metrics_server_run(metrics_server_t* server) {
357 if (!server) return;
358
359 server->running = 1;
360 printf("[METRICS] Server started, accepting connections...\n");
361
362 while (server->running) {
363 fd_set read_fds;
364 FD_ZERO(&read_fds);
365 FD_SET(server->listen_fd, &read_fds);
366
367 struct timeval tv;
368 tv.tv_sec = 1;
369 tv.tv_usec = 0;
370
371 int ret = select((int)server->listen_fd + 1, &read_fds, NULL, NULL, &tv);
372 if (ret < 0) {
373 if (
374#ifdef _WIN32
375 WSAGetLastError() == WSAEINTR
376#else
377 errno == EINTR
378#endif
379 ) {
380 continue;
381 }
382 break;
383 }
384
385 if (ret == 0) continue;
386
387 /* 接受新连接 */
388 if (FD_ISSET(server->listen_fd, &read_fds)) {
389 struct sockaddr_in client_addr;
390 socklen_t addr_len = sizeof(client_addr);
391 SOCKET client_fd = accept(server->listen_fd,
392 (struct sockaddr*)&client_addr, &addr_len);
393 if (client_fd != INVALID_SOCKET) {
394 /* 设置 TCP_NODELAY */
395 int opt = 1;
396 setsockopt(client_fd, IPPROTO_TCP, TCP_NODELAY,
397 (const char*)&opt, sizeof(opt));
398
399 /* 处理单个连接（阻塞读取，但 select 已保证可读） */
400 metrics_handle_connection(server, client_fd);
401 }
402 }
403 }
